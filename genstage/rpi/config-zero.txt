# zero config.txt — read by the BCM2837 GPU firmware (start.elf) from the
# FAT32 boot partition. Serves the FULL-stage3 'zero' profile; the tiny
# track's twin is config-zero2w.txt (same hardware, same knobs).

# the Zero 2 W is ARMv8 — run the 64-bit kernel8.img
arm_64bit=1
kernel=kernel8.img

# don't wait for a rainbow splash
disable_splash=1

# headless board: leave the 512 MB to the CPU
gpu_mem=16

# serial console on GPIO 14/15 (3V3 TTL, 115200 8N1)
enable_uart=1
# hand the PL011 back to the header: serial0 becomes ttyAMA0 with a fixed
# clock. Without this, serial0 is the mini-UART, whose baud follows the core
# clock.
dtoverlay=disable-bt

# SPI0 (CE0) for the Waveshare 2.13" V4 e-Paper HAT (SSD1680); creates
# /dev/spidev0.0, which /usr/bin/eink213 talks to.
dtparam=spi=on

# USB controller in peripheral/OTG mode, so g_ether can make the board appear
# to the machine at the other end of the cable as a USB network adapter.
# Plug into the USB port (the inner one), not PWR IN.
#
# The overlay's own default is dr_mode="otg" (checked in the compiled
# dwc2.dtbo, and the base DTB's usb@7e980000 node sets no dr_mode, no
# id-gpios and no usb-role-switch — so the role comes from the dwc2
# controller's own ID-pin sensing, nothing else). The ADAPTER therefore picks
# the role at plug time, and both uses of the port work without editing this
# file:
#
#   plain micro-USB data cable to a PC   ID floating -> peripheral role,
#                                        g_ether binds, usb0 comes up at the
#                                        address the profile pins
#   micro-USB OTG adapter + WiFi stick   ID pulled  -> host role, the COMFAST
#                                        low         CF-922AC (MT7612U)
#                                                    enumerates, mt76x2u binds
#
# It senses the ADAPTER, not the device: a WiFi stick on a plain micro-B cable
# leaves ID floating and the port stays a gadget. A cheap adapter that does not
# actually pull ID low behaves the same way — that is the usual cause of "host
# mode does not work", not the overlay.
#
# The two roles are mutually exclusive: the Zero 2 W has exactly ONE data port
# (the second micro-USB, PWR IN, has no data lines to the SoC), so whenever the
# CF-922AC is plugged in there is no usb0 and no ssh over the gadget link —
# reach the board over WiFi or the serial console below. Power from PWR IN in
# that case; the stick draws up to ~450 mA on TX.
#
# To take the adapter out of the decision, pin the role explicitly — at the
# cost of being able to hot-swap between the two:
#   dtoverlay=dwc2,dr_mode=host        USB WiFi always, gadget never
#   dtoverlay=dwc2,dr_mode=peripheral  gadget always, USB WiFi never
dtoverlay=dwc2
