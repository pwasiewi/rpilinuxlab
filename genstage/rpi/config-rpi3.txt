# config.txt — Raspberry Pi 3 (BCM2837/BCM2710, Cortex-A53)
# Installed on the FAT32 boot partition by: xstage arm64|tiny image (BOARD=rpi3)
#
# Boot chain on this SoC (like the Zero 2 W, unlike the EEPROM-booting Pi 4/400):
#   boot ROM -> bootcode.bin -> start.elf + fixup.dat -> config.txt
#   -> kernel8.img + bcm2710-rpi-3-b.dtb (auto-selected) + cmdline.txt
# All of those files come from sys-kernel/raspberrypi-image (xstage <t> rpi).

arm_64bit=1             # the Pi 3 is ARMv8 — run the 64-bit kernel8.img
kernel=kernel8.img
disable_splash=1

enable_uart=1           # serial console on GPIO 14/15 (3V3 TTL, 115200 8N1)
dtoverlay=disable-bt    # hand the PL011 back to the header: serial0 becomes
                        # ttyAMA0 with a fixed clock. Without this, serial0 is
                        # the mini-UART, whose baud follows the core clock.

# emit HDMI even when the firmware fails to detect the display at power-on
# (monitor plugged in late, KVM, quirky EDID) — part of dark-HDMI debugging
hdmi_force_hotplug=1
